#!/usr/bin/env python3
"""Apple native package notice corpus validator and XCFramework notice finalizer (DEF-0030).

validate-corpus     check the source notice corpus (bindings/native/third-party) against the native lock:
                    every third-party lock package is classified exactly once, roles agree with the
                    recorded Apple graph, texts exist with their indexed bytes and nothing else is present.
finalize            add the notice closure to a freshly built, notice-free XCFramework: a byte copy of
                    its compilation parent (native-build-parent.json), NOTICE.md, the indexed corpus under
                    third-party/ and packaging.json listing every other member; then validate the result.
validate-framework  recheck a source or packed copy. The compilation parent's SHA256 is trusted external
                    input: the parent copy and packaging record shipped inside the artifact never certify
                    themselves, and notice bytes are compared with the source corpus.

Every check is an explicit raise (behavior is identical under python -O). No network or subprocess.

  apple_notices.py validate-corpus --corpus DIR --lock Cargo.lock
  apple_notices.py finalize --framework DIR --parent native-build.json --corpus DIR --lock Cargo.lock
  apple_notices.py validate-framework --framework DIR --corpus DIR --lock Cargo.lock --expected-parent-sha256 HEX
"""
import argparse
import hashlib
import json
import os
from pathlib import Path, PurePosixPath
import re
import sys
import tomllib

HEX64 = re.compile(r'[0-9a-f]{64}')
APPLE_TARGETS = ('x86_64-apple-darwin', 'aarch64-apple-ios', 'aarch64-apple-ios-sim', 'x86_64-apple-ios')
CRATE_ROLES = ('linked_rust', 'build_tool', 'build_native_c_source')
BUILD_ROLES = ('build_tool', 'build_native_c_source')
APPLE_ROLES = CRATE_ROLES + ('linked_native_c', 'unused_source_not_linked', 'os_framework_not_redistributed',
                             'toolchain_aggregate')
ANDROID_ONLY_ROLES = ('kotlin_runtime',)
TEXT_ROLES = CRATE_ROLES + ('linked_native_c', 'toolchain_aggregate')
OS_PROVIDERS = {'CommonCrypto': ('_CC',), 'Security': ('_Sec', '_kSec')}
TOOLCHAIN_AGGREGATES = ('rust-std',)
PARENT_COPY = 'native-build-parent.json'
PACKAGING = 'packaging.json'
NOTICE = 'NOTICE.md'
INDEX = 'index.json'
CORPUS_DIR = 'third-party'


class NoticeError(Exception):
    def __init__(self, reason, message):
        super().__init__(f'{reason}: {message}')
        self.reason = reason


def _sha(data):
    return hashlib.sha256(data).hexdigest()


def _safe_relative(value, what):
    if not isinstance(value, str) or not value or '\\' in value or value.startswith('/') or '//' in value:
        raise NoticeError('unsafe-path', f'{what} must be a relative POSIX path: {value!r}')
    if any(part in ('', '.', '..') for part in PurePosixPath(value).parts):
        raise NoticeError('unsafe-path', f'{what} must not contain empty, "." or ".." segments: {value!r}')
    return value


def _files(directory):
    """Every regular file below directory, keyed by POSIX relative path; symbolic links are rejected."""
    found = {}
    for dirpath, dirnames, filenames in os.walk(directory):
        for name in dirnames + filenames:
            if os.path.islink(os.path.join(dirpath, name)):
                raise NoticeError('unexpected', f'symbolic link: {os.path.join(dirpath, name)}')
        for name in filenames:
            path = os.path.join(dirpath, name)
            found[PurePosixPath(os.path.relpath(path, directory)).as_posix()] = Path(path)
    return found


def _load_json(path, reason, what):
    try:
        return json.loads(Path(path).read_bytes())
    except (OSError, ValueError) as error:
        raise NoticeError(reason, f'{what} is not readable JSON: {error}')


def _lock_packages(lock_bytes):
    try:
        lock = tomllib.loads(lock_bytes.decode('utf-8'))
    except (UnicodeDecodeError, tomllib.TOMLDecodeError) as error:
        raise NoticeError('lock-mismatch', f'lock is not TOML: {error}')
    packages = {}
    for package in lock.get('package', []):
        if 'source' not in package:
            continue  # First-party workspace/path crate; no third-party attribution.
        key = (package['name'], package['version'])
        if key in packages:
            raise NoticeError('lock-mismatch', f'duplicate lock package {key}')
        packages[key] = package
    return packages


def _check_text_entries(package, corpus_files):
    entries = package.get('files')
    if not isinstance(entries, list):
        raise NoticeError('missing', f'{package.get("name")!r} needs a files list')
    if package['role'] in TEXT_ROLES and not entries:
        raise NoticeError('missing', f'{package["name"]!r} ({package["role"]}) has no license text')
    declared = []
    for entry in entries:
        if not isinstance(entry, dict) or not isinstance(entry.get('origin'), dict):
            raise NoticeError('unclassified', f'{package["name"]!r} text entry needs file, sha256 and origin')
        relative = _safe_relative(entry.get('file'), f'{package["name"]} text path')
        if relative in (NOTICE, INDEX):
            raise NoticeError('unsafe-path', f'{relative} is reserved')
        if not isinstance(entry.get('sha256'), str) or not HEX64.fullmatch(entry['sha256']):
            raise NoticeError('unclassified', f'{relative} needs a lowercase SHA256')
        if relative not in corpus_files:
            raise NoticeError('missing', f'indexed text is missing: {relative}')
        if _sha(corpus_files[relative].read_bytes()) != entry['sha256']:
            raise NoticeError('changed', f'indexed text bytes changed: {relative}')
        declared.append(relative)
    return declared


def _check_provider(package):
    name = package['name']
    if name not in OS_PROVIDERS:
        raise NoticeError('provider-role', f'{name!r} is not a known OS-provided Apple framework')
    if package.get('files'):
        raise NoticeError('provider-role', f'{name} is provided by the OS and must not claim redistributed texts')
    evidence = package.get('evidence')
    if not isinstance(evidence, list) or not evidence:
        raise NoticeError('provider-role', f'{name} needs per-library imported-symbol evidence')
    for item in evidence:
        if not isinstance(item, dict) or not isinstance(item.get('library'), str) or not HEX64.fullmatch(str(item.get('sha256'))):
            raise NoticeError('provider-role', f'{name} evidence needs library and sha256')
        symbols = item.get('imported_symbols')
        if not isinstance(symbols, list) or not symbols or not all(isinstance(s, str) and s.startswith(OS_PROVIDERS[name]) for s in symbols):
            raise NoticeError('provider-role', f'{name} evidence for {item["library"]} needs its imported {name} symbols')
        if item.get('libtomcrypt_symbols') != 0:
            raise NoticeError('wrong-platform', f'{item["library"]} records LibTomCrypt symbols; Apple links OS CommonCrypto')
    return {item['library']: item['sha256'] for item in evidence}


def _check_toolchain(package):
    name = package['name']
    if 'ndk' in name.lower() or 'android' in name.lower():
        raise NoticeError('wrong-platform', f'{name!r} is an Android toolchain aggregate')
    if name not in TOOLCHAIN_AGGREGATES:
        raise NoticeError('unclassified', f'{name!r} is not a known Apple toolchain aggregate')
    inputs = package.get('verified_link_inputs')
    if not isinstance(inputs, list) or not inputs:
        raise NoticeError('missing', f'{name} needs verified link inputs')
    for item in inputs:
        if not isinstance(item, dict) or not isinstance(item.get('path'), str) or not HEX64.fullmatch(str(item.get('sha256'))):
            raise NoticeError('unclassified', f'{name} link input needs target, path and sha256')
        if item.get('target') not in APPLE_TARGETS or 'android' in item['path'].lower():
            raise NoticeError('wrong-platform', f'{name} link input is not for an Apple target: {item.get("target")!r} {item["path"]}')


def validate_corpus(corpus, lock):
    corpus, lock = Path(corpus), Path(lock)
    if not corpus.is_dir():
        raise NoticeError('missing', f'corpus directory does not exist: {corpus}')
    if not lock.is_file():
        raise NoticeError('missing', f'lock does not exist: {lock}')
    corpus_files = _files(corpus)
    for required in (NOTICE, INDEX):
        if required not in corpus_files:
            raise NoticeError('missing', f'corpus needs {required}')
    index = _load_json(corpus / INDEX, 'unclassified', 'notice index')
    if not isinstance(index, dict) or index.get('format') != 1:
        raise NoticeError('unclassified', 'notice index format must be 1')
    if index.get('platform') != 'apple':
        raise NoticeError('wrong-platform', f'notice index platform must be apple, not {index.get("platform")!r}')
    if index.get('project_distribution_license', 'absent') is not None:
        raise NoticeError('unclassified', 'project_distribution_license must be recorded as null (none chosen)')
    if tuple(index.get('apple_targets') or ()) != APPLE_TARGETS:
        raise NoticeError('wrong-platform', f'apple_targets must be exactly {list(APPLE_TARGETS)}')
    notice = index.get('notice')
    if not isinstance(notice, dict) or notice.get('file') != NOTICE:
        raise NoticeError('unclassified', 'notice index must bind NOTICE.md')
    if _sha(corpus_files[NOTICE].read_bytes()) != notice.get('sha256'):
        raise NoticeError('changed', 'NOTICE.md bytes changed')

    lock_bytes = lock.read_bytes()
    if not isinstance(index.get('lock'), dict) or index['lock'].get('sha256') != _sha(lock_bytes):
        raise NoticeError('lock-mismatch', f'notice index is bound to different lock bytes than {lock}')
    locked = _lock_packages(lock_bytes)

    graph = index.get('graph')
    if not isinstance(graph, dict) or not isinstance(graph.get('normal'), list) or not isinstance(graph.get('build'), list):
        raise NoticeError('unclassified', 'notice index needs the recorded Apple normal and build graph')
    normal, build = set(graph['normal']), set(graph['build'])
    if normal & build:
        raise NoticeError('wrong-role', f'graph entries are both normal and build: {sorted(normal & build)}')

    packages = index.get('packages')
    if not isinstance(packages, list) or not packages:
        raise NoticeError('unclassified', 'notice index needs packages')
    names, declared, crates, providers, roles = set(), [], {}, {}, {}
    for package in packages:
        if not isinstance(package, dict) or not isinstance(package.get('name'), str):
            raise NoticeError('unclassified', 'every package needs a name')
        name, role = package['name'], package.get('role')
        if name in names:
            raise NoticeError('duplicate', f'package {name!r} is listed more than once')
        names.add(name)
        if role in ANDROID_ONLY_ROLES:
            raise NoticeError('wrong-platform', f'{name!r} role {role!r} is Android-only')
        if role not in APPLE_ROLES:
            raise NoticeError('unclassified', f'{name!r} has unknown role {role!r}')
        roles[name] = role
        if role in CRATE_ROLES:
            key = (name, package.get('version'))
            entry = locked.get(key)
            if entry is None or entry.get('source') != package.get('source') or entry.get('checksum') != package.get('checksum'):
                raise NoticeError('lock-mismatch', f'{key} source/checksum does not match the lock')
            label = f'{key[0]} {key[1]}'
            if role == 'linked_rust' and label not in normal:
                raise NoticeError('wrong-role', f'{label} is linked_rust but not on the Apple normal graph')
            if role in BUILD_ROLES and label not in build:
                raise NoticeError('wrong-role', f'{label} is {role} but not on the Apple build graph')
            crates[label] = key
        elif role == 'os_framework_not_redistributed':
            providers[name] = _check_provider(package)
        elif role == 'toolchain_aggregate':
            _check_toolchain(package)
        declared += _check_text_entries(package, corpus_files)

    for package in packages:
        role = package['role']
        if role in ('linked_native_c', 'unused_source_not_linked'):
            if roles.get(package.get('via')) != 'build_native_c_source':
                raise NoticeError('unclassified', f'{package["name"]!r} needs via naming its build_native_c_source crate')
            if 'tomcrypt' in package['name'].lower() and role != 'unused_source_not_linked':
                raise NoticeError('wrong-platform', 'LibTomCrypt is not linked on Apple (OS CommonCrypto provider)')
            if role == 'unused_source_not_linked' and package['files']:
                raise NoticeError('wrong-role', f'{package["name"]!r} is unused source and must not claim attribution texts')
    missing_providers = sorted(set(OS_PROVIDERS) - set(providers))
    if missing_providers:
        raise NoticeError('provider-role', f'OS crypto providers must be recorded separately: {missing_providers}')

    unattributed = sorted((normal | build) - set(crates))
    if unattributed:
        raise NoticeError('unclassified', f'Apple graph members without attribution: {unattributed}')
    excluded = index.get('excluded_lock_packages')
    if not isinstance(excluded, list):
        raise NoticeError('unclassified', 'notice index needs excluded_lock_packages')
    excluded_keys = set()
    for item in excluded:
        if not isinstance(item, dict) or not isinstance(item.get('reason'), str) or not item['reason']:
            raise NoticeError('unclassified', 'every excluded lock package needs name, version and reason')
        key = (item.get('name'), item.get('version'))
        if f'{key[0]} {key[1]}' in normal | build:
            raise NoticeError('wrong-role', f'{key} is on the Apple graph and cannot be excluded')
        if key not in locked or key in excluded_keys:
            raise NoticeError('lock-mismatch', f'excluded package {key} is not a unique lock package')
        excluded_keys.add(key)
    unclassified = sorted(set(locked) - set(crates.values()) - excluded_keys)
    if unclassified:
        raise NoticeError('unclassified', f'lock packages neither attributed nor excluded: {unclassified}')

    if len(declared) != len(set(declared)):
        raise NoticeError('duplicate', 'a corpus text is indexed more than once')
    expected = set(declared) | {NOTICE, INDEX}
    extra = sorted(set(corpus_files) - expected)
    if extra:
        raise NoticeError('unexpected', f'unindexed corpus files: {extra[:5]}')
    return {'result': 'corpus-valid', 'index_sha256': _sha(corpus_files[INDEX].read_bytes()),
            'notice_sha256': notice['sha256'], 'lock_sha256': _sha(lock_bytes), 'texts': len(declared),
            'packages': len(packages), 'excluded_lock_packages': len(excluded_keys), 'provider_evidence': providers}


def _notice_members(corpus):
    """Framework-relative notice members and their bytes, derived from the validated source corpus."""
    members = {}
    for relative, path in _files(corpus).items():
        members[NOTICE if relative == NOTICE else f'{CORPUS_DIR}/{relative}'] = path.read_bytes()
    return members


def _native_expected(framework, parent_bytes):
    try:
        parent = json.loads(parent_bytes)
    except ValueError as error:
        raise NoticeError('parent-mismatch', f'compilation parent is not JSON: {error}')
    recorded = parent.get('artifact_files_sha256') if isinstance(parent, dict) else None
    if not isinstance(recorded, dict) or not recorded:
        raise NoticeError('parent-mismatch', 'compilation parent records no native artifact files')
    prefix = framework.name + '/'
    native = {}
    for key, digest in recorded.items():
        if not isinstance(key, str) or not key.startswith(prefix) or not HEX64.fullmatch(str(digest)):
            raise NoticeError('native-mismatch', f'parent member is not inside {framework.name}: {key!r}')
        relative = _safe_relative(key[len(prefix):], 'native member')
        if relative in (PARENT_COPY, PACKAGING, NOTICE) or relative.split('/')[0] == CORPUS_DIR:
            raise NoticeError('native-mismatch', f'parent claims a packaging member as native: {relative}')
        native[relative] = digest
    return native


def _check_provider_binding(native, providers):
    for name, libraries in providers.items():
        for library, digest in libraries.items():
            if native.get(library) != digest:
                raise NoticeError('stale-provider-evidence', f'{name} evidence for {library} does not match the shipped library bytes')
        missing = sorted(k for k in native if k.endswith('.a') and k not in libraries)
        if missing:
            raise NoticeError('stale-provider-evidence', f'{name} evidence does not cover shipped libraries: {missing}')


def _packaging_record(framework, parent_bytes, members):
    record = {'format': 1, 'framework': framework.name, 'parent': {'file': PARENT_COPY, 'sha256': _sha(parent_bytes)},
              'notice_index': {'file': f'{CORPUS_DIR}/{INDEX}', 'sha256': members[f'{CORPUS_DIR}/{INDEX}']},
              'members': members}
    return (json.dumps(record, indent=2, sort_keys=True) + '\n').encode('utf-8')


def finalize_framework(framework, parent, corpus, lock):
    framework, parent = Path(framework), Path(parent)
    corpus_summary = validate_corpus(corpus, lock)
    if not framework.is_dir() or not parent.is_file():
        raise NoticeError('missing', 'framework directory and compilation parent record must exist')
    parent_bytes = parent.read_bytes()
    native = _native_expected(framework, parent_bytes)
    actual = _files(framework)
    extra = sorted(set(actual) - set(native))
    if extra:
        raise NoticeError('unexpected', f'finalize needs the notice-free native framework; found {extra[:5]}')
    for relative, digest in native.items():
        if relative not in actual or _sha(actual[relative].read_bytes()) != digest:
            raise NoticeError('native-mismatch', f'native member differs from its compilation parent: {relative}')
    _check_provider_binding(native, corpus_summary['provider_evidence'])

    notices = _notice_members(Path(corpus))
    for relative, data in notices.items():
        target = framework.joinpath(*PurePosixPath(relative).parts)
        target.parent.mkdir(parents=True, exist_ok=True)
        target.write_bytes(data)
    (framework / PARENT_COPY).write_bytes(parent_bytes)
    members = dict(native)
    members[PARENT_COPY] = _sha(parent_bytes)
    members.update({relative: _sha(data) for relative, data in notices.items()})
    packaging = _packaging_record(framework, parent_bytes, dict(sorted(members.items())))
    (framework / PACKAGING).write_bytes(packaging)
    summary = validate_framework(framework, corpus, lock, _sha(parent_bytes))
    summary['result'] = 'finalized'
    return summary


def validate_framework(framework, corpus, lock, expected_parent_sha256):
    framework = Path(framework)
    if not isinstance(expected_parent_sha256, str) or not HEX64.fullmatch(expected_parent_sha256):
        raise NoticeError('invalid-input', 'expected parent SHA256 must be 64 lowercase hex characters')
    corpus_summary = validate_corpus(corpus, lock)
    if not framework.is_dir():
        raise NoticeError('missing', f'framework does not exist: {framework}')
    actual = _files(framework)
    if PACKAGING not in actual:
        raise NoticeError('missing-packaging', f'{framework.name} has no {PACKAGING} (notice-free artifact)')
    if PARENT_COPY not in actual:
        raise NoticeError('missing-packaging', f'{framework.name} has no {PARENT_COPY}')
    parent_bytes = actual[PARENT_COPY].read_bytes()
    if _sha(parent_bytes) != expected_parent_sha256:
        raise NoticeError('parent-mismatch', f'{PARENT_COPY} is not the trusted compilation parent {expected_parent_sha256}')
    native = _native_expected(framework, parent_bytes)
    notices = {relative: _sha(data) for relative, data in _notice_members(Path(corpus)).items()}
    expected = dict(native)
    expected[PARENT_COPY] = expected_parent_sha256
    expected.update(notices)

    unexpected = sorted(set(actual) - set(expected) - {PACKAGING})
    if unexpected:
        raise NoticeError('unexpected', f'members outside the native and notice closure: {unexpected[:5]}')
    for relative, digest in sorted(expected.items()):
        kind = 'native-mismatch' if relative in native else 'notice-mismatch'
        if relative not in actual:
            raise NoticeError(kind, f'missing member: {relative}')
        if _sha(actual[relative].read_bytes()) != digest:
            raise NoticeError(kind, f'member bytes differ from the trusted source: {relative}')
    _check_provider_binding(native, corpus_summary['provider_evidence'])

    packaging_bytes = actual[PACKAGING].read_bytes()
    if packaging_bytes != _packaging_record(framework, parent_bytes, dict(sorted(expected.items()))):
        raise NoticeError('packaging-mismatch', f'{PACKAGING} does not list exactly the expected members and hashes')
    return {'result': 'framework-valid', 'framework': framework.name, 'parent_sha256': expected_parent_sha256,
            'packaging_sha256': _sha(packaging_bytes), 'index_sha256': corpus_summary['index_sha256'],
            'members': len(expected), 'native_members': len(native), 'notice_members': len(notices)}


def main(argv=None):
    parser = argparse.ArgumentParser(description=__doc__, formatter_class=argparse.RawDescriptionHelpFormatter)
    sub = parser.add_subparsers(dest='command', required=True)
    c = sub.add_parser('validate-corpus')
    c.add_argument('--corpus', required=True)
    c.add_argument('--lock', required=True)
    f = sub.add_parser('finalize')
    for name in ('--framework', '--parent', '--corpus', '--lock'):
        f.add_argument(name, required=True)
    v = sub.add_parser('validate-framework')
    for name in ('--framework', '--corpus', '--lock', '--expected-parent-sha256'):
        v.add_argument(name, required=True)
    args = parser.parse_args(argv)
    try:
        if args.command == 'validate-corpus':
            summary = validate_corpus(args.corpus, args.lock)
        elif args.command == 'finalize':
            summary = finalize_framework(args.framework, args.parent, args.corpus, args.lock)
        else:
            summary = validate_framework(args.framework, args.corpus, args.lock, args.expected_parent_sha256)
    except NoticeError as error:
        print(f'REJECTED {error}', file=sys.stderr)
        return 2
    except OSError as error:
        print(f'REJECTED io-error: {error}', file=sys.stderr)
        return 2
    print(json.dumps(summary, sort_keys=True))
    return 0


if __name__ == '__main__':
    sys.exit(main())
