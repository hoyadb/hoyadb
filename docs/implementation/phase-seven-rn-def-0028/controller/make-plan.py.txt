"""Trusted activation controller step: derive the DEF-0028 assembler input plan from frozen approval records.

Hashes always come from the approval records (ADR-0035 artifact index, 8a87389 bundle/responses, compiler
bundle 364a4201, root's provenance-checked metadata lists). Where an approval record lacks a size, the
size is taken from the already-verified local bytes; the record's hash still decides integrity in the
assembler. No byte is fetched: a missing local input stops plan generation. Explicit checks only.
Usage: make-plan.py PLAN_OUT.json
"""
from pathlib import Path
import hashlib
import json
import sys

repo_root = Path(__file__).resolve().parents[3]
out = Path(sys.argv[1])
if out.exists():
    raise SystemExit('plan output must not exist')
sha = lambda data, a='sha256': hashlib.new(a, data).hexdigest()


def fail(message):
    raise SystemExit('PLAN REJECTED: ' + message)


def layout(coordinate, name):
    group, artifact, version = coordinate.split(':')[:3]
    return f"{group.replace('.', '/')}/{artifact}/{version}/{name}"


def rel(path):
    path = Path(path).resolve()
    if repo_root.resolve() not in path.parents:
        fail(f'input outside the workspace root: {path}')
    return str(path.relative_to(repo_root.resolve()))


entries = {}


def add(path, kind, source, sha256, sha512, size, url, identity):
    if size is None:
        size = (repo_root / source).stat().st_size
    entry = {'path': path, 'kind': kind, 'source': {'root': 'workspace', 'path': source}, 'size': size,
             'sha256': sha256, 'sha512': sha512 or None, 'provenance': {'url': url, 'identity': identity}}
    previous = entries.get(path)
    if previous is None:
        entries[path] = entry
        return
    if (previous['sha256'], previous['size']) != (sha256, size):
        fail(f'conflicting approved bytes for {path}')
    if previous['kind'] != kind:
        fail(f'conflicting kinds for {path}')
    # The same approved bytes are recorded by more than one approval record: one entry naming every record.
    previous['provenance']['identity'] += ' | ' + identity
    previous['sha512'] = previous['sha512'] or sha512 or None


# 1. ADR-0035 artifact index (androidDebug, nativeBuildTools, kotlinSdkRuntime) from verified local copies.
index = json.loads((repo_root / 'docs/implementation/phase-seven-dependencies/native/artifact-index.json').read_text())
verified = {x['coordinate'] + '|' + Path(x['path']).name: repo_root / x['path']
            for x in json.loads((repo_root / 'target/phase-seven-rn/maven/verified.json').read_text())['verified']}
review = repo_root / 'target/phase-seven-dependencies/native-source-review'
fetched = repo_root / 'target/phase-seven-rn/android/fetched'
for entry in index['artifacts']:
    if not {'androidDebug', 'nativeBuildTools', 'kotlinSdkRuntime'} & set(entry['configurations']):
        continue
    coordinate, name = entry['coordinate'], entry['name']
    candidates = [verified.get(coordinate + '|' + name), review / 'artifact-byte-review' / (coordinate.replace(':', '__') + '__' + name),
                  fetched / layout(coordinate, name)]
    source = next((p for p in candidates if p is not None and p.is_file()), None)
    if source is None:
        fail(f'no verified local bytes for approved {coordinate} {name}')
    pin = entry.get('sha256') or entry.get('sha256_pin')
    add(layout(coordinate, name), 'binary', rel(source), pin, entry.get('sha512'), entry.get('size'), entry['qualified_url'],
        f'ADR-0035 artifact-index {coordinate} {name}')

# 2. Plugin addendum 8a87389 (JARs and recorded repository responses).
addendum_docs = repo_root / 'docs/implementation/phase-seven-plugin-addendum'
archive = repo_root / 'target/phase-seven-plugin-addendum/archive-review'
for item in json.loads((addendum_docs / 'bundle.json').read_text())['additional_artifacts']:
    name = item['url'].rsplit('/', 1)[1]
    add(layout(item['coordinate'], name), 'binary', rel(archive / layout(item['coordinate'], name)), item['sha256'], None, item['bytes'],
        item['url'], f'8a87389 additional artifact {item["coordinate"]}')
for response in json.loads((addendum_docs / 'repository-responses.json').read_text()):
    name = response['url'].rsplit('/', 1)[1]
    add(layout(response['coordinate'], name), 'metadata', rel(archive / layout(response['coordinate'], name)), response['sha256'], None, None,
        response['url'], f'8a87389 repository response {response["coordinate"]} {response["type"]}')

# 3. Approved compiler bundle (5e771f2 / index 2182049f / bundle 364a4201).
packet = repo_root / 'docs/implementation/phase-seven-kotlin-compiler-addendum'
bundle_bytes = (packet / 'bundle.json').read_bytes()
if sha(bundle_bytes) != '364a420165b6f4301d90b23179e43f26c78de6e75b9f1059d7c97f30fc3cafcf':
    fail('compiler bundle differs from approved 364a4201')
if sha((packet / 'files-sha256.json').read_bytes()) != '2182049f4dd8b8623b04ff4c45dbf7e2fb5d79f171f34693e1384493e2b674e5':
    fail('compiler packet index differs from approved 2182049f')
compiler = json.loads(bundle_bytes)
marker = compiler['marker']
add(layout(marker['coordinate'], marker['packet_bytes'].rsplit('/', 1)[1]), 'metadata', rel(packet / marker['packet_bytes']), marker['sha256'],
    None, marker['bytes'], 'https://repo.maven.apache.org/maven2/' + layout(marker['coordinate'], marker['packet_bytes'].rsplit('/', 1)[1]),
    'compiler bundle 364a4201 marker')
quarantine = repo_root / 'target/phase-seven-rn/compiler-review/closure'
for comp in compiler['additional_jar_components'] + compiler['metadata_only_components']:
    for f in comp['files']:
        add(layout(comp['coordinate'], f['name']), 'binary' if f['name'].endswith('.jar') else 'metadata',
            rel(quarantine / comp['coordinate'].replace(':', '__') / f['name']), f['sha256'], f['sha512'], f['bytes'], f['url'],
            f'compiler bundle 364a4201 {comp["coordinate"]} {f["name"]}')

# 4. Root's provenance-checked captured metadata.
provenance = {}
for listing in ['metadata-provenance.json', 'selected-metadata-provenance.json', 'selected-patched-metadata-provenance.json']:
    for item in json.loads((review / 'maven' / listing).read_text()):
        recorded = provenance.get(item['file'])
        if recorded is None:
            provenance[item['file']] = item
        elif recorded['sha256'] != item['sha256'] or recorded['url'] != item['url']:
            fail('conflicting captured metadata provenance for ' + item['file'])
for relative, item in sorted(provenance.items()):
    add(relative, 'metadata', rel(review / 'maven' / relative), item['sha256'], None, None, item['url'], 'root captured metadata provenance')

plan = {'format': 1, 'roots': {'workspace': str(repo_root.resolve())}, 'entries': [entries[k] for k in sorted(entries)]}
out.write_text(json.dumps(plan, indent=2) + '\n')
kinds = {k: sum(1 for e in entries.values() if e['kind'] == k) for k in ('binary', 'metadata')}
print('HOYA_PLAN', json.dumps({'entries': len(entries), 'kinds': kinds, 'plan_sha256': sha(out.read_bytes())}))
