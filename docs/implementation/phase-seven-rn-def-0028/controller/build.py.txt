def digest(path):
    return hashlib.sha256(path.read_bytes()).hexdigest()


def tree(root):
    return {str(p.relative_to(root)): digest(p) for p in sorted(root.rglob('*')) if p.is_file() and not p.is_symlink()}


def step(name, argv, cwd, env=None, timeout=3600):
    entry = {'name': name, 'argv': argv, 'cwd': str(cwd)}
    record['steps'].append(entry)
    started = time.monotonic()
    try:
        with (out / f'{name}.txt').open('w') as log:
            p = subprocess.run(argv, cwd=cwd, env=env, stdout=log, stderr=subprocess.STDOUT, timeout=timeout)
        entry['exit_code'] = p.returncode
    except subprocess.TimeoutExpired:
        entry['timed_out'] = True
        raise
    finally:
        entry['elapsed_s'] = round(time.monotonic() - started, 3)
        save()
    if p.returncode:
        raise RuntimeError(f'{name} exit {p.returncode}')



def approved_union():
    idx = json.loads((repo_root / 'docs/implementation/phase-seven-dependencies/native/artifact-index.json').read_text())['artifacts']
    union = {a['coordinate'] for a in idx}
    union |= set(json.loads((repo_root / 'docs/implementation/phase-seven-plugin-addendum/bundle.json').read_text())['additional_plugin_coordinates'])
    compiler = json.loads((repo_root / 'docs/implementation/phase-seven-kotlin-compiler-addendum/bundle.json').read_text())
    union |= {c['coordinate'] for c in compiler['additional_jar_components'] + compiler['metadata_only_components']}
    union.add(compiler['marker']['coordinate'])
    return union


def compare_graphs(logs):
    union = approved_union()
    result = {'per_log': {}, 'unapproved': {}}
    for log in logs:
        selected, unresolved = set(), []
        for line in log.read_text().splitlines():
            m = re.search(r'--- (.+)$', line)
            if not m:
                continue
            item = re.sub(r' \((\*|c|n)\)$', '', m.group(1)).strip()
            if item.startswith('project ') or item.endswith('FAILED'):
                if item.endswith('FAILED'):
                    unresolved.append(item)
                continue
            if ' -> ' in item:
                left, right = item.split(' -> ', 1)
                parts = left.split(':')
                coordinate = f'{parts[0]}:{parts[1]}:{right.split()[0]}'
            else:
                parts = item.split(':')
                if len(parts) < 3:
                    continue
                coordinate = ':'.join(parts[:3]).split()[0]
            selected.add(coordinate)
        unknown = sorted(c for c in selected if c not in union)
        result['per_log'][log.name] = {'selected': sorted(selected), 'unresolved': unresolved, 'unapproved': unknown}
        if unknown or unresolved:
            result['unapproved'][log.name] = unknown + unresolved
    return result
